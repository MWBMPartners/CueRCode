Allow: /

# 🚫 Disallow private/API paths
# 🔒 NOTE: /install/ is intentionally NOT listed here (F-020). A robots.txt

#    to scanners. The directory is instead blocked at the HTTP layer by
#    install/.htaccess (Require all denied) and should be deleted post-setup.
Disallow: /api/
Disallow: /error/
